     3→const API_BASE = '/api'
     4→
     5→class ApiClient {
     6→  private token: string | null = null
     7→
     8→  setToken(token: string | null) {
     9→    this.token = token
    10→  }
    11→
    12→  getToken(): string | null {
    13→    return this.token
    14→  }
    15→
    16→  private async request(path: string, options: RequestInit = {}) {
    17→    const headers: Record<string, string> = {
    18→      'Content-Type': 'application/json',
    19→      ...(options.headers as Record<string, string> || {}),
    20→    }
    21→    if (this.token) {
    22→      headers['Authorization'] = `Bearer ${this.token}`
    23→    }
    24→    const res = await fetch(`${API_BASE}${path}`, { ...options, headers })
    25→    if (!res.ok) {
    26→      const error = await res.json().catch(() => ({ error: 'Error de conexión' }))
    27→      throw new Error(error.error || `Error ${res.status}`)
    28→    }
    29→    return res.json()
    30→  }

   271→  // WhatsApp — SIEMPRE vía /api/whatsapp/daemon-proxy con JWT (auditoría Antigravity
   272→  // críticos #3/#4: antes el frontend hablaba directo al puerto 3002, sin auth;
   273→  // el envío manual devolvía 401 y las conversaciones eran públicamente legibles).
   274→  private async proxyGet(path: string) {
   275→    const headers: Record<string, string> = {}
   276→    if (this.token) headers['Authorization'] = `Bearer ${this.token}`
   277→    const res = await fetch(`/api/whatsapp/daemon-proxy?path=${encodeURIComponent(path)}`, { headers, cache: 'no-store' })
   278→    return res.json()
   279→  }
   280→
   281→  private async proxyPost(path: string, data: Record<string, unknown> = {}) {
   282→    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   283→    if (this.token) headers['Authorization'] = `Bearer ${this.token}`
   284→    const res = await fetch('/api/whatsapp/daemon-proxy', {
   285→      method: 'POST',
   286→      headers,
   287→      body: JSON.stringify({ path, ...data }),
   288→    })
   289→    return res.json()
   290→  }
   291→

   295→    const res = await fetch('/api/whatsapp/daemon-proxy', {
   296→      method: 'PUT',
   297→      headers,
   298→      body: JSON.stringify({ path, ...data }),

   300→    return res.json()
   301→  }

   541→    const usp = new URLSearchParams()
   542→    if (params.q) usp.set('q', params.q)
   543→    if (params.clientId) usp.set('clientId', params.clientId)
   544→    if (params.k) usp.set('k', String(params.k))

   546→    return this.request(`/memory/vector${qs ? `?${qs}` : ''}`)
   547→  }

   648→  /** (A3-08, auditoría 2026-11-11) Exporta cualquier entidad a CSV con el JWT
   649→   *  en el HEADER y descarga por blob: el token JAMÁS viaja en la URL (queda en
   650→   *  logs de servidor, historial del navegador y cabeceras Referer). */
   651→  async downloadExportCsv(type: string, filename: string) {
   652→    const headers: Record<string, string> = {}
   653→    if (this.token) {
   654→      headers['Authorization'] = `Bearer ${this.token}`
   655→    }
   656→    const res = await fetch(`${API_BASE}/export/csv?type=${encodeURIComponent(type)}`, { headers })
   657→    if (!res.ok) {
   658→      const error = await res.json().catch(() => ({ error: 'Error de conexión' }))
   659→      throw new Error(error.error || `Error ${res.status}`)
   660→    }
   661→    const blob = await res.blob()
   662→    const url = URL.createObjectURL(blob)
   663→    const a = document.createElement('a')
   664→    a.href = url
   665→    a.download = filename
   666→    a.click()
   667→    URL.revokeObjectURL(url)
   668→  }
