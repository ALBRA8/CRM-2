   309→interface FocusResponse {

   314→// href del focus → id de vista del shell (app-shell.tsx, patrón de navegación
   315→// de esta página: setView). /pipeline se renderiza como vista 'opportunities';
   316→// el resto mapea 1:1 con las páginas existentes en src/app.
   317→const FOCUS_HREF_TO_VIEW: Record<string, string> = {
   318→  '/inbox': 'inbox',
   319→  '/automations': 'automations',
   320→  '/calendar': 'calendar',
   321→  '/quotes': 'quotes',

   328→  inbox: { icon: Inbox, chip: 'bg-emerald-50 text-emerald-600' },

   337→  2: { label: 'Media', className: 'bg-amber-50 text-amber-700 border-0' },

   339→}

   341→/**
   342→ * U2 — Sección accionable "Qué te toca ahora": items priorizados por
   343→ * GET /api/dashboard/focus. Estados propios (carga, vacío, error) para no
   344→ * acoplarse a la carga de las métricas ni alterar las secciones existentes.
   345→ */
   346→function FocusSection() {
   347→  const { setView } = useAppStore()
   348→  const [items, setItems] = useState<FocusItem[]>([])
   349→  const [loading, setLoading] = useState(true)
   350→  const [error, setError] = useState(false)
   351→
   352→  useEffect(() => {
   353→    loadFocus()
   354→  }, [])
   355→
   356→  const loadFocus = async () => {
   357→    try {
   358→      setLoading(true)
   359→      setError(false)
   360→      const data = await api.getDashboardFocus() as FocusResponse
   361→      setItems(Array.isArray(data?.items) ? data.items : [])
   362→    } catch {
   363→      setError(true)
   364→    } finally {
   365→      setLoading(false)
   366→    }
   367→  }
   368→
   369→  const navigate = (href: string) => {
   370→    setView(FOCUS_HREF_TO_VIEW[href] ?? href.replace(/^\//, ''))
   371→  }
   372→
   373→  return (
   374→    <motion.div
   375→      initial={{ opacity: 0, y: 12 }}
   376→      animate={{ opacity: 1, y: 0 }}
   377→      transition={{ duration: 0.3 }}
   378→    >
   379→      <Card className="border-0 shadow-sm bg-white border-l-4 border-l-emerald-500">
   380→        <CardHeader className="pb-3">
   381→          <CardTitle className="text-lg font-semibold text-slate-900 flex items-center gap-2">
   382→            <ListChecks className="w-5 h-5 text-emerald-600" />
   383→            Qué te toca ahora
   384→          </CardTitle>
   385→          <p className="text-xs text-slate-400">Pendientes accionables ordenados por prioridad</p>
   386→        </CardHeader>
   387→        <CardContent>
   388→          {loading ? (
   389→            <div className="space-y-2">
   390→              {[1, 2, 3].map((i) => <Skeleton key={i} className="h-12 rounded-lg" />)}
   391→            </div>
   392→          ) : error ? (
   393→            <div className="py-4 text-center">
   394→              <p className="text-sm text-slate-400">No se pudo cargar el focus de acciones.</p>
   395→              <Button onClick={loadFocus} variant="outline" size="sm" className="mt-3">
   396→                Reintentar
   397→              </Button>
   398→            </div>
   399→          ) : items.length > 0 ? (
   400→            <div className="space-y-1">
   401→              {items.map((item) => {
   402→                const style = focusTypeStyles[item.type]
   403→                const badge = priorityBadge[item.priority]
   404→                const Icon = style.icon
   405→                return (
   406→                  <button
   407→                    key={item.id}
   408→                    onClick={() => navigate(item.href)}
   409→                    className="w-full flex items-center gap-3 p-2.5 rounded-lg hover:bg-slate-50 transition-colors text-left group"
   410→                  >
   411→                    <div className={`w-8 h-8 rounded-lg flex items-center justify-center flex-shrink-0 ${style.chip}`}>
   412→                      <Icon className="w-4 h-4" />
   413→                    </div>
   414→                    <div className="min-w-0 flex-1">
   415→                      <p className="text-sm font-medium text-slate-900 truncate">{item.title}</p>
   416→                      <p className="text-xs text-slate-500 truncate">{item.detail}</p>
   417→                    </div>
   418→                    <Badge variant="outline" className={`text-[10px] flex-shrink-0 ${badge.className}`}>
   419→                      {badge.label}
   420→                    </Badge>
   421→                    <ChevronRight className="w-4 h-4 text-slate-300 flex-shrink-0 group-hover:text-slate-500 transition-colors" />
   422→                  </button>
   423→                )
   424→              })}
   425→            </div>
   426→          ) : (
   427→            <div className="py-6 text-center text-slate-400 text-sm">
   428→              <ListChecks className="w-10 h-10 mx-auto mb-2 text-slate-200" />
   429→              Todo al día. No hay nada urgente.
   430→            </div>
   431→          )}
   432→        </CardContent>
   433→      </Card>
   434→    </motion.div>
   435→  )
   436→}

   459→        api.getDashboard(),

   466→        console.error('Dashboard error:', dashboardResult.reason)
   467→        setError('Error al cargar los datos del dashboard')

   470→      if (suggestionsResult.status === 'fulfilled') {
   471→        const sugData = suggestionsResult.value as SuggestionsData
   472→        setSuggestions(sugData.suggestions ?? [])

   474→      // Suggestions failure is non-critical, silently ignore
   475→    } catch {
   476→      setError('Error de conexión al cargar el dashboard')

   478→        description: 'No se pudieron cargar los datos del dashboard',
