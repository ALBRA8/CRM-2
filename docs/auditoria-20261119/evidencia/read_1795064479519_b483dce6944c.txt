     3→const API_BASE = '/api'

     5→class ApiClient {
     6→  private token: string | null = null
     7→
     8→  setToken(token: string | null) {
     9→    this.token = token
    10→  }
    11→
    12→  getToken(): string | null {
    13→    return this.token
    14→  }
    15→
    16→  private async request(path: string, options: RequestInit = {}) {
    17→    const headers: Record<string, string> = {
    18→      'Content-Type': 'application/json',
    19→      ...(options.headers as Record<string, string> || {}),
    20→    }
    21→    if (this.token) {
    22→      headers['Authorization'] = `Bearer ${this.token}`
    23→    }
    24→    const res = await fetch(`${API_BASE}${path}`, { ...options, headers })
    25→    if (!res.ok) {
    26→      const error = await res.json().catch(() => ({ error: 'Error de conexión' }))
    27→      throw new Error(error.error || `Error ${res.status}`)
    28→    }
    29→    return res.json()
    30→  }

   271→  // WhatsApp — SIEMPRE vía /api/whatsapp/daemon-proxy con JWT (auditoría Antigravity

   274→  private async proxyGet(path: string) {
   275→    const headers: Record<string, string> = {}
   276→    if (this.token) headers['Authorization'] = `Bearer ${this.token}`
   277→    const res = await fetch(`/api/whatsapp/daemon-proxy?path=${encodeURIComponent(path)}`, { headers, cache: 'no-store' })
   278→    return res.json()
   279→  }
   280→
   281→  private async proxyPost(path: string, data: Record<string, unknown> = {}) {
   282→    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   283→    if (this.token) headers['Authorization'] = `Bearer ${this.token}`
   284→    const res = await fetch('/api/whatsapp/daemon-proxy', {
   285→      method: 'POST',
   286→      headers,
   287→      body: JSON.stringify({ path, ...data }),
   288→    })
   289→    return res.json()
   290→  }

   672→  /** (A3-08, auditoría 2026-11-11) Exporta cualquier entidad a CSV con el JWT
   673→   *  en el HEADER y descarga por blob: el token JAMÁS viaja en la URL (queda en
   674→   *  logs de servidor, historial del navegador y cabeceras Referer). */
   675→  async downloadExportCsv(type: string, filename: string) {
   676→    const headers: Record<string, string> = {}
   677→    if (this.token) {
   678→      headers['Authorization'] = `Bearer ${this.token}`
   679→    }
   680→    const res = await fetch(`${API_BASE}/export/csv?type=${encodeURIComponent(type)}`, { headers })
   681→    if (!res.ok) {
   682→      const error = await res.json().catch(() => ({ error: 'Error de conexión' }))
   683→      throw new Error(error.error || `Error ${res.status}`)
   684→    }
   685→    const blob = await res.blob()
   686→    const url = URL.createObjectURL(blob)
   687→    const a = document.createElement('a')
   688→    a.href = url
   689→    a.download = filename
   690→    a.click()
   691→    URL.revokeObjectURL(url)
   692→  }
